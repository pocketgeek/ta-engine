cmake_minimum_required(VERSION 3.20)
# Packaging only: never relink, strip or otherwise mutate signed executables.
if(NOT TA_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "TA_VERSION must be a release version")
endif()
project(ta-engine VERSION ${TA_VERSION} LANGUAGES NONE)
# This installer carries x64 binaries; LANGUAGES NONE does not detect pointer size.
set(CMAKE_SIZEOF_VOID_P 8)
if(NOT IS_DIRECTORY "${TA_SIGNED_PAYLOAD}")
  message(FATAL_ERROR "TA_SIGNED_PAYLOAD must point at the verified x64 release payload")
endif()
foreach(name taclient.exe taserver.exe cartographer.exe
             STREAMING-LICENSES.txt EDITOR-FONT-LICENSE.txt CHAT-FONT-LICENSE.txt TITLE-FONT-LICENSE.txt ACME-LICENSES.txt OPUS-LICENSE.txt)
  if(NOT EXISTS "${TA_SIGNED_PAYLOAD}/${name}")
    message(FATAL_ERROR "Missing installer payload: ${name}")
  endif()
  install(FILES "${TA_SIGNED_PAYLOAD}/${name}" DESTINATION .)
endforeach()
set(TA_PACKAGE_CARTOGRAPHER ON)
include(installer.cmake)
include(CPack)
